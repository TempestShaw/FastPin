cmake_minimum_required(VERSION 3.24)
project(FastPin VERSION 0.1.0 LANGUAGES CXX)
set(FASTPIN_VERSION "${PROJECT_VERSION}" CACHE STRING "Displayed release version")
set(FASTPIN_BUILD_VERSION "${PROJECT_VERSION}" CACHE STRING "Increasing Sparkle build version")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
    # Keep the executable, DLLs and qt.conf together at the MSIX root.
    set(CMAKE_INSTALL_BINDIR ".")
endif()
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version for the chosen Qt binaries")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network)

add_library(fastpin_core STATIC capture.cpp pin.cpp ocr.cpp annotation.cpp toolbar.cpp texteditor.cpp history.cpp hotkey.cpp shortcut.cpp onboarding.cpp translation.cpp translationcard.cpp)
qt_add_resources(fastpin_core dictionary PREFIX "/dictionary" BASE "dictionary" FILES dictionary/cedict.tsv.qz)
if(APPLE)
    enable_language(OBJCXX)
    include(FetchContent)
    FetchContent_Declare(sparkle
        URL https://github.com/sparkle-project/Sparkle/releases/download/2.10.0/Sparkle-2.10.0.tar.xz
        URL_HASH SHA256=c2bf58aa8387266ac179357b1415d6f2635f044da8be41042af32425dae6da0c
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(sparkle)
    add_library(Sparkle SHARED IMPORTED)
    set_target_properties(Sparkle PROPERTIES
        IMPORTED_LOCATION "${sparkle_SOURCE_DIR}/Sparkle.framework/Sparkle"
        FRAMEWORK TRUE)
    target_sources(fastpin_core PRIVATE platform_mac.mm settings_mac.mm)
    set_source_files_properties(platform_mac.mm settings_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(fastpin_core PUBLIC Sparkle "-framework AppKit" "-framework AVFoundation" "-framework QuartzCore")
    target_link_libraries(fastpin_core PUBLIC "-framework Vision" "-framework ApplicationServices")
elseif(WIN32)
    target_sources(fastpin_core PRIVATE platform_windows.cpp)
    target_link_libraries(fastpin_core PUBLIC uiautomationcore ole32 dwmapi windowsapp uuid sapi)
    target_compile_options(fastpin_core PUBLIC /utf-8)
endif()
target_link_libraries(fastpin_core PUBLIC Qt6::Widgets Qt6::Network)
target_include_directories(fastpin_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

qt_add_executable(FastPin WIN32 MACOSX_BUNDLE main.cpp)
target_compile_definitions(FastPin PRIVATE FASTPIN_VERSION="${FASTPIN_VERSION}")
target_link_libraries(FastPin PRIVATE fastpin_core)
qt_add_resources(FastPin assets PREFIX "/brand" BASE "assets" FILES assets/app-icon.png assets/tray-icon.png)
if(APPLE)
    set_target_properties(FastPin PROPERTIES BUILD_RPATH "@executable_path/../Frameworks")
    add_custom_command(TARGET FastPin POST_BUILD
        COMMAND /usr/bin/ditto "${sparkle_SOURCE_DIR}/Sparkle.framework" "$<TARGET_BUNDLE_DIR:FastPin>/Contents/Frameworks/Sparkle.framework")
    target_sources(FastPin PRIVATE assets/app-icon.icns)
    set_source_files_properties(assets/app-icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_link_libraries(fastpin_core PUBLIC "-framework Carbon" "-framework CoreGraphics")
    set_target_properties(FastPin PROPERTIES
        MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.tempestshaw.fastpin"
        MACOSX_BUNDLE_BUNDLE_NAME "FastPin"
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in)
elseif(WIN32)
    target_sources(FastPin PRIVATE assets/app-icon.rc)
    target_include_directories(FastPin PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/assets)
    target_link_libraries(fastpin_core PUBLIC user32)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(fastpin_tests tests.cpp)
    target_link_libraries(fastpin_tests PRIVATE fastpin_core Qt6::Test)
    target_compile_definitions(fastpin_tests PRIVATE FASTPIN_APPLICATION="$<TARGET_FILE:FastPin>")
    add_dependencies(fastpin_tests FastPin)
    add_test(NAME native_core COMMAND fastpin_tests)
    set_tests_properties(native_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(fastpin_controller_tests controller_tests.cpp)
    target_link_libraries(fastpin_controller_tests PRIVATE fastpin_core Qt6::Test)
    target_compile_definitions(fastpin_controller_tests PRIVATE FASTPIN_VERSION="${FASTPIN_VERSION}")
    add_test(NAME shortcut_lifecycle COMMAND fastpin_controller_tests)
    set_tests_properties(shortcut_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(APPLE)
        qt_add_executable(fastpin_mac_settings_tests mac_settings_tests.mm)
        set_source_files_properties(mac_settings_tests.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        target_link_libraries(fastpin_mac_settings_tests PRIVATE fastpin_core Qt6::Test)
        add_test(NAME mac_settings COMMAND fastpin_mac_settings_tests)
        qt_add_executable(fastpin_mac_updater_smoke mac_updater_smoke.mm)
        set_source_files_properties(mac_updater_smoke.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        target_link_libraries(fastpin_mac_updater_smoke PRIVATE fastpin_core)
        set_target_properties(fastpin_mac_updater_smoke PROPERTIES BUILD_RPATH "@executable_path/../Frameworks")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME mac_updater COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run-macos-updater.py
            $<TARGET_FILE:fastpin_mac_updater_smoke> ${sparkle_SOURCE_DIR})
        set_tests_properties(mac_updater PROPERTIES TIMEOUT 120)
    endif()
    qt_add_executable(fastpin_benchmark benchmark.cpp)
    target_link_libraries(fastpin_benchmark PRIVATE fastpin_core)
endif()

install(TARGETS FastPin BUNDLE DESTINATION . RUNTIME DESTINATION .)
if(APPLE)
    set(notice_dir "FastPin.app/Contents/Resources")
else()
    set(notice_dir ".")
endif()
install(FILES LICENSE.md GETTING_STARTED.md NOTICE.md DESTINATION ${notice_dir})
install(DIRECTORY licenses DESTINATION ${notice_dir})
install(FILES public/dmg/install-preview.png DESTINATION ${notice_dir}/public/dmg)
install(FILES public/windows/extract-zip.png public/windows/developer-mode.png public/windows/ready-to-launch.png
        DESTINATION ${notice_dir}/public/windows)
if(WIN32)
    set(FASTPIN_WINDOWS_PUBLISHER "CN=FastPin" CACHE STRING "MSIX publisher certificate subject")
    string(REPLACE "&" "&amp;" FASTPIN_WINDOWS_PUBLISHER_XML "${FASTPIN_WINDOWS_PUBLISHER}")
    string(REPLACE "\"" "&quot;" FASTPIN_WINDOWS_PUBLISHER_XML "${FASTPIN_WINDOWS_PUBLISHER_XML}")
    string(REPLACE "<" "&lt;" FASTPIN_WINDOWS_PUBLISHER_XML "${FASTPIN_WINDOWS_PUBLISHER_XML}")
    string(REPLACE ">" "&gt;" FASTPIN_WINDOWS_PUBLISHER_XML "${FASTPIN_WINDOWS_PUBLISHER_XML}")
    configure_file(AppxManifest.xml.in AppxManifest.xml @ONLY)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/AppxManifest.xml DESTINATION .)
endif()
if(WIN32)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
    include(InstallRequiredSystemLibraries)
    if(NOT CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS)
        message(FATAL_ERROR "MSVC redistributable DLLs are required for an app-local MSIX runtime")
    endif()
    qt_generate_deploy_app_script(TARGET FastPin OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR
        DEPLOY_TOOL_OPTIONS --no-compiler-runtime --no-translations --no-system-d3d-compiler --no-system-dxc-compiler
                            --skip-plugin-types generic,networkinformation)
else()
    qt_generate_deploy_app_script(TARGET FastPin OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
endif()
install(SCRIPT ${deploy_script})
